Pixels au hasard

; remplit l'écran de couleurs au hasard
        ld hl,#4000
loop:   in a,(#FE)
        ld (hl),a
        inc hl
        ld a,h
        cp #44
        jr nz,loop
        halt

Une touche, une couleur

; cliquer sur l'écran, puis taper des touches
loop:   in a,(#FF)
        or a
        jr z,loop
        ld (#4210),a
        jr loop
